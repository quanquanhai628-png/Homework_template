const Engine = Matter.Engine;
const World = Matter.World;
const Bodies = Matter.Bodies;
let engine;
let world;
let balls = [];
let boxes = [];
let ground;
let platform;
function setup() {
createCanvas(800, 600);
// Physics Engine
engine = Engine.create();
world = engine.world;
// Gravity
engine.gravity.y = 1;
// Fixed ground
ground = Bodies.rectangle(400, 580, 800, 30, {
isStatic: true,
restitution: 0.3,
friction: 0.8
});
// Tilted platform
platform = Bodies.rectangle(400, 430, 500, 20, {
isStatic: true,
angle: radians(-8),
restitution: 0.6,
friction: 0.6
});
World.add(world, [ground, platform]);
// Create balls
for (let i = 0; i < 8; i++) {
let x = 180 + i * 65;
let y = 80 + (i % 3) * 40;
let ball = Bodies.circle(x, y, 22 + (i % 3) * 6, {
restitution: 0.7,
friction: 0.3,
density: 0.001 + i * 0.0003
});
balls.push(ball);
World.add(world, ball);
}
// Create boxes
for (let i = 0; i < 4; i++) {
let x = 250 + i * 90;
let y = 40;
let box = Bodies.rectangle(x, y, 45 + i * 8, 45 + i * 5, {
restitution: 0.4,
friction: 0.6,
density: 0.002
});
boxes.push(box);
World.add(world, box);
}
// Start physics
Engine.run(engine);
}
function draw() {
background(245);
// Title
fill(30);
noStroke();
textSize(20);
textAlign(CENTER);
text("DYNAMIC BALANCE", width / 2, 35);
// Draw platform
drawBody(platform);
// Draw ground
drawBody(ground);
// Draw balls
for (let ball of balls) {
drawBall(ball);
}
// Draw boxes
for (let box of boxes) {
drawBox(box);
}
}
function drawBody(body) {
push();
translate(body.position.x, body.position.y);
rotate(body.angle);
rectMode(CENTER);
fill(45);
noStroke();
rect(0, 0, body.bounds.max.x - body.bounds.min.x,
body.bounds.max.y - body.bounds.min.y);
pop();
}
function drawBall(ball) {
push();
translate(ball.position.x, ball.position.y);
rotate(ball.angle);
fill(80);
noStroke();
ellipseMode(CENTER);
ellipse(0, 0, ball.circleRadius * 2);
pop();
}
function drawBox(box) {
push();
translate(box.position.x, box.position.y);
rotate(box.angle);
rectMode(CENTER);
fill(150);
noStroke();
let w = box.bounds.max.x - box.bounds.min.x;
let h = box.bounds.max.y - box.bounds.min.y;
rect(0, 0, w, h);
pop();
}
